create or alter procedure company_mast_insert_update_del(
@action varchar(max),
@comp_id bigint ,
@comp_code varchar(25),
@comp_name varchar(50),
@comp_short_name varchar(6),
@comp_type varchar(50),
@comp_Desc varchar(50),
@cin_number varchar(25),
@gst_number varchar(25),
@pan_number varchar(12),
@contperson_name varchar(50),
@contact_email varchar(50),
@conatact_phone Varchar(12),
@address_line1 Varchar(max),
@address_line2 Varchar(max),
@city_id Bigint,
@pincode varchar(6),
@is_Active bit,
@created_date date,
@updated_date date,
@logo_path varchar(50),
@user_id bigint,
)
as
begin

declare @ErrorNumber int, @ErrorProcedure nvarchar(128), @ErrorLine int, @ErrorMessage nvarchar(max);


if @action='insert'
begin
	begin try
		begin transaction
			insert into company_mast(
			comp_code,
			comp_name,
			comp_short_name,
			comp_type,
			comp_Desc,
			cin_number,
			gst_number,
			pan_number,
			contperson_name,
			contact_email,
			conatact_phone,
			address_line1,
			address_line2,
			city_id,
			pincode,
			is_Active,
			created_date,
			updated_date,
			logo_path,
			user_id)

			values(
			@comp_code,
			@comp_name,
			@comp_short_name,
			@comp_type,
			@comp_Desc,
			@cin_number,
			@gst_number,
			@pan_number,
			@contperson_name,
			@contact_email,
			@conatact_phone,
			@address_line1,
			@address_line2,
			@city_id,
			@pincode,
			@is_Active,
			@created_date,
			@updated_date,
			@logo_path,
			@user_id
			)
			
		commit transaction;
	end try
		begin catch
			IF XACT_STATE() <> 0
			rollback transaction;

			set @ErrorNumber = ERROR_NUMBER();
			set @ErrorProcedure = ERROR_PROCEDURE();
			set @ErrorLine = ERROR_LINE();
			set @ErrorMessage = ERROR_MESSAGE();

			exec ErrorLog_Ins 
				 @ErrorNumber,
				 @ErrorProcedure,
				 @ErrorLine,
				 @ErrorMessage;	
		end catch
end



if @action='delete'
begin	
		declare @rec_count bigint
	begin try
		begin transaction;
		set @rec_count = (select 
            (select count(comp_id) from user_details where comp_id = @comp_id) +
            (select count(comp_id) from inward_mast where comp_id = @comp_id) +
            (select count(comp_id) from inward_return where comp_id = @comp_id) +
            (select count(comp_id) from salesinvoice_mast where comp_id = @comp_id) +
            (select count(comp_id) from salesinvoicedetails where comp_id = @comp_id) +
            (select count(comp_id) from challan_mast where comp_id = @comp_id) +
            (select count(comp_id) from receipt_mast where comp_id = @comp_id) +
            (select count(comp_id) from receipt_details where comp_id = @comp_id) +
            (select count(comp_id) from purchaseinvoice_mast where comp_id = @comp_id) +
            (select count(comp_id) from PurchaseInvoice_Details where comp_id = @comp_id) +
            (select count(comp_id) from dailyconsumption_mast where comp_id = @comp_id) +
            (select count(comp_id) from employee_mast where comp_id = @comp_id) +
            (select count(company_id) from employee_payslip where comp_id = @comp_id) +
            (select count(comp_id) from payment_mast where comp_id = @comp_id) +
            (select count(comp_id) from payment_details where comp_id = @comp_id) +			
		);
		
		if @rec_count > 0
		begin
			rollback transaction;
			return;
		end

		delete from company_mast where comp_id=@comp_id
		commit transaction;
	
	end try
		begin catch
			If XACT_STATE() <> 0
			rollback transaction;

			set @ErrorNumber = ERROR_NUMBER();
			set @ErrorProcedure = ERROR_PROCEDURE();
			set @ErrorLine = ERROR_LINE();
			set @ErrorMessage = ERROR_MESSAGE();

			exec ErrorLog_Ins
			@ErrorNumber,
			@ErrorProcedure,
			@ErrorLine,
			@ErrorMessage			
		end catch
end



if @action='update'
begin
	begin try
		begin transaction
			update company_mast 
			set 
			comp_code=@comp_code,
			comp_name=@comp_name,
			comp_short_name=@comp_short_name,
			comp_type=@comp_type,
			comp_Desc=@comp_Desc,
			cin_number=@cin_number,
			gst_number=@gst_number,
			pan_number=@pan_number,
			contperson_name=@contperson_name,
			contact_email=@contact_email,
			conatact_phone=@conatact_phone,
			address_line1=@address_line1,
			address_line2=@address_line2,
			city_id=@city_id,
			pincode=@pincode,
			is_Active=@is_Active,
			created_date=@created_date,
			updated_date=@updated_date,
			logo_path=@logo_path,
			user_id=@user_id
			where comp_id=@comp_id,
				
		if @@ROWCOUNT = 0
		begin
			rollback transaction;
			return;
		end

		commit transaction;
	end try
		begin catch
			IF XACT_STATE() <> 0 
			rollback transaction;

			set @ErrorNumber = ERROR_NUMBER();
			set @ErrorProcedure = ERROR_PROCEDURE();
			set @ErrorLine = ERROR_LINE();
			set @ErrorMessage = ERROR_MESSAGE();

			exec ErrorLog_Ins
			@ErrorNumber,
			@ErrorProcedure,
			@ErrorLine,
			@ErrorMessage			
		end catch
end



if @action='selectall'
begin
	begin try
		select 
		comp_id,
		comp_code,
		comp_name,
		comp_short_name,
		comp_type,
		comp_Desc,
		cin_number,
		gst_number,
		pan_number,
		contperson_name,
		contact_email,
		conatact_phone,
		address_line1,
		address_line2,
		city_id,
		pincode,
		is_Active,
		created_date,
		updated_date,
		logo_path,
		user_id from company_mast;
		
	end try
		begin catch
			set @ErrorNumber = ERROR_NUMBER();
			set @ErrorProcedure = ERROR_PROCEDURE();
			set @ErrorLine = ERROR_LINE();
			set @ErrorMessage = ERROR_MESSAGE();

			exec ErrorLog_Ins
			@ErrorNumber,
			@ErrorProcedure,
			@ErrorLine,
			@ErrorMessage
		end catch
end



if @action='selectone'
begin
	begin try
		select 
		comp_id,
		comp_code,
		comp_name,
		comp_short_name,
		comp_type,
		comp_Desc,
		cin_number,
		gst_number,
		pan_number,
		contperson_name,
		contact_email,
		conatact_phone,
		address_line1,
		address_line2,
		city_id,
		pincode,
		is_Active,
		created_date,
		updated_date,
		logo_path,
		user_id from company_mast where comp_id=@comp_id ;
	
	end try
	begin catch
		set @ErrorNumber = ERROR_NUMBER();
		set @ErrorProcedure = ERROR_PROCEDURE();
		set @ErrorLine = ERROR_LINE();
		set @ErrorMessage = ERROR_MESSAGE();

		exec ErrorLog_Ins
		@ErrorNumber,
		@ErrorProcedure,
		@ErrorLine,
		@ErrorMessage
	end catch
end



if @action='companylist'
begin
	begin try
		select comp_id,comp_name from company_mast;
	end try
	begin catch
		
		set @ErrorNumber = ERROR_NUMBER();
		set @ErrorProcedure = ERROR_PROCEDURE();
		set @ErrorLine = ERROR_LINE();
		set @ErrorMessage = ERROR_MESSAGE();

		exec ErrorLog_Ins
		@ErrorNumber,
		@ErrorProcedure,
		@ErrorLine,
		@ErrorMessage			
	end catch
end


end
