create or alter procedure product_mast_insert_update_del(
@action varchar(max),
@product_id bigint=0,
@prodtype_id bigint=0,
@brand_id bigint=0,
@hsn_id bigint=0,
@unit_id bigint=0,
@product_name varchar(50)='',
@product_desc varchar(50)='',
@rate Decimal(12,2)=0,
@opening_stock decimal(10,2)=0,
@Purchase decimal(10,2)=0,
@Sales decimal(10,2)=0,
@[Return] decimal(10,2)=0,
@current_stock decimal(10,2)=0,
@Reorder_Threshold decimal(10,2)=0,
@Reorder_Desc varchar(50)='',
@created_date date=null,
@updated_date date=null,
@user_id bigint=0
)
as

begin

declare @ErrorNumber int, @ErrorProcedure nvarchar(128), @ErrorLine int, @ErrorMessage nvarchar(max);

if @action='insert'
begin
	begin try
		begin transaction
			insert into product_mast(prodtype_id,brand_id,hsn_id,unit_id,product_name,product_desc,rate,opening_stock,Purchase,Sales,Return,current_stock,Reorder_Threshold,Reorder_Desc,created_date,updated_date,user_id)
			values(@prodtype_id,@brand_id,@hsn_id,@unit_id,@product_name,@product_desc,@rate,@opening_stock,@Purchase,@Sales,@Return,@current_stock,@Reorder_Threshold,@Reorder_Desc,@created_date,@updated_date,@user_id);
		commit transaction;
	end try
		begin catch
			IF XACT_STATE() <> 0
			rollback transaction;

			set @ErrorNumber = ERROR_NUMBER();
			set @ErrorProcedure = ERROR_PROCEDURE();
			set @ErrorLine = ERROR_LINE();
			set @ErrorMessage = ERROR_MESSAGE();

			exec ErrorLog_Ins 
				 @ErrorNumber,
				 @ErrorProcedure,
				 @ErrorLine,
				 @ErrorMessage;	
		end catch
end



if @action='delete'
begin	
		declare @rec_count bigint
	begin try
		begin transaction;
		set @rec_count = (select 
            (
                select count(product_id) from inward_mast where product_id = @product_id
            ) +
            (
                select count(product_id) from inward_return where product_id = @product_id
            ) +
            (
                select count(product_id) from PurchaseInvoice_Details where product_id = @product_id
            ) +
            (
                select count(product_id) from salesinvoicedetails where product_id = @product_id
            ) +
            (
                select count(product_id) from dailyconsumption_mast where product_id = @product_id
            )
		);
		
		if @rec_count > 0
			begin
				rollback transaction;
				return;
			end

		delete from product_mast where product_id=@product_id;
		commit transaction;
		
	end try
		begin catch
			If XACT_STATE() <> 0
			rollback transaction;

			set @ErrorNumber = ERROR_NUMBER();
			set @ErrorProcedure = ERROR_PROCEDURE();
			set @ErrorLine = ERROR_LINE();
			set @ErrorMessage = ERROR_MESSAGE();

			exec ErrorLog_Ins
			@ErrorNumber,
			@ErrorProcedure,
			@ErrorLine,
			@ErrorMessage			
		end catch
end



if @action='update'
begin
	begin try
		begin transaction
			update product_mast
			set
			prodtype_id=@prodtype_id,
			brand_id=@brand_id,
			hsn_id=@hsn_id,
			unit_id=@unit_id,
			product_name=@product_name,
			product_desc=@product_desc,
			rate=@rate,
			opening_stock=@opening_stock,
			Purchase=@Purchase,
			Sales=@Sales,
			Return=@Return,
			current_stock=@current_stock,
			Reorder_Threshold=@Reorder_Threshold,
			Reorder_Desc=@Reorder_Desc,
			created_date=@created_date,
			updated_date=@updated_date,
			user_id=@user_id
			where product_id=@product_id,
			
		if @@ROWCOUNT = 0
		begin
			rollback transaction;
			return;
		end

		commit transaction;
	end try
		begin catch
			IF XACT_STATE() <> 0 
			rollback transaction;

			set @ErrorNumber = ERROR_NUMBER();
			set @ErrorProcedure = ERROR_PROCEDURE();
			set @ErrorLine = ERROR_LINE();
			set @ErrorMessage = ERROR_MESSAGE();

			exec ErrorLog_Ins
			@ErrorNumber,
			@ErrorProcedure,
			@ErrorLine,
			@ErrorMessage			
		end catch
end



if @action='selectall'
begin
	begin try
		a.select product_id,
		a.prodtype_id,
		b.prodtype_name
		a.brand_id,
		c.brand_name
		a.hsn_id,
		d.hsn_code,
		a.unit_id,
		e.unit_name
		a.product_name,
		a.product_desc,
		a.rate,
		a.opening_stock,
		a.Purchase,
		a.Sales,
		a.Return,
		a.current_stock,
		a.Reorder_Threshold,
		a.Reorder_Desc,
		a.created_date,
		a.updated_date,
		a.user_id 
		a.from product_mast a
		a.inner join prodtype_master b on a.prodtype_id=b.prodtype_id
		a.inner join brand_mast c on a.brand_id=c.brand_id
		a.inner join hsn_mast d on a.hsn_id=d.hsn_id
		a.inner join unit_mast e on a.unit_id=e.unit_id;
		
	end try
		begin catch
			set @ErrorNumber = ERROR_NUMBER();
			set @ErrorProcedure = ERROR_PROCEDURE();
			set @ErrorLine = ERROR_LINE();
			set @ErrorMessage = ERROR_MESSAGE();

			exec ErrorLog_Ins
			@ErrorNumber,
			@ErrorProcedure,
			@ErrorLine,
			@ErrorMessage
		end catch
end



if @action='selectone'
begin
	begin try
		select product_id,prodtype_id,brand_id,hsn_id,unit_id,product_name,product_desc,rate,opening_stock,Purchase,Sales,Return,current_stock,Reorder_Threshold,Reorder_Desc,created_date,updated_date,user_id from product_mast where product_id=@product_id;
	end try
		begin catch
			set @ErrorNumber = ERROR_NUMBER();
			set @ErrorProcedure = ERROR_PROCEDURE();
			set @ErrorLine = ERROR_LINE();
			set @ErrorMessage = ERROR_MESSAGE();

			exec ErrorLog_Ins
			@ErrorNumber,
			@ErrorProcedure,
			@ErrorLine,
			@ErrorMessage
		end catch
end



if @action = 'drop down'
begin
	begin try
		select product_id,product_name from product_mast;
	end try
	begin catch
		
		set @ErrorNumber = ERROR_NUMBER();
		set @ErrorProcedure = ERROR_PROCEDURE();
		set @ErrorLine = ERROR_LINE();
		set @ErrorMessage = ERROR_MESSAGE();

		exec ErrorLog_Ins
		@ErrorNumber,
		@ErrorProcedure,
		@ErrorLine,
		@ErrorMessage			
	end catch
end


end
